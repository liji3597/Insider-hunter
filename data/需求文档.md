结合你的“内幕猎手”需求，我们比文档中的标准表多加几个字段。

```python
from sqlalchemy import Column, String, Integer, Numeric, DateTime, Boolean, create_engine
from sqlalchemy.ext.declarative import declarative_base

Base = declarative_base()

class Market(Base):
    __tablename__ = "markets"
    id = Column(Integer, primary_key=True, index=True)
    slug = Column(String, unique=True, index=True) # 如 "trump-nomination"
    condition_id = Column(String)
    yes_token_id = Column(String)
    no_token_id = Column(String)
    category = Column(String) # "Politics", "Business" - 用于筛选垂类
    active = Column(Boolean, default=True)

class Trade(Base):
    __tablename__ = "trades"
    tx_hash = Column(String, primary_key=True)
    log_index = Column(Integer, primary_key=True)
    market_slug = Column(String, index=True) # 方便关联
    maker = Column(String, index=True)       # 大户地址
    side = Column(String)   # BUY/SELL
    outcome = Column(String) # YES/NO
    amount_usd = Column(Numeric(18, 2)) # 成交金额 (核心过滤指标)
    timestamp = Column(DateTime, index=True)
    
    # 🌟 内幕猎手专属字段
    is_whale = Column(Boolean, default=False) # 是否大单 (>1000U)
    news_checked = Column(Boolean, default=False) # Agent 是否已分析过

```

### 第二步：抓取垂类市场 (Day 1 中午)

**逻辑：** 我们不抓所有市场，只抓“政治”类且活跃的市场。
**参考文档：** 阶段二 - 任务 A (Market Discovery)。

编写 `src/indexer/discovery.py`:

```python
import requests

def fetch_politics_markets():
    # Gamma API: 筛选 closed=false, limit=100, sort=volume
    url = "https://gamma-api.polymarket.com/markets?active=true&closed=false&limit=50&tag_id=1" 
    # tag_id=1 通常是 Politics，或者通过 /tags 接口查询
    
    resp = requests.get(url).json()
    markets = []
    for m in resp:
        # 只存我们需要的数据
        markets.append({
            "slug": m['slug'],
            "condition_id": m['condition_id'],
            "yes_token_id": m['clobTokenIds'][0], # 假设0是YES，需按文档校验
            "no_token_id": m['clobTokenIds'][1],
            "category": "Politics"
        })
    return markets

```

def listen_loop():
    print("开始监听链上交易...")
    event_filter = w3.eth.filter({
        "address": EXCHANGE_ADDRESS,
        "topics": ["0x..."] # OrderFilled 的 topic hash (查文档或计算)
    })
    
    while True:
        for event in event_filter.get_new_entries():
            try:
                # 1. 解码交易
                trade_data = decode_log(event) 
                
                # 2. 匹配已存的政治市场 (通过 token_id)

                    (Market.yes_token_id == trade_data['token_id']) |

                if market:
